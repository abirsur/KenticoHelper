from qwen_vl_utils import process_vision_info

# Define a custom collator function
def collate_fn(examples):
    texts = [processor.apply_chat_template(example["messages"], tokenize=False) for example in examples]
    image_inputs = [process_vision_info(example["messages"])[0] for example in examples]

    batch = processor(text=texts, images=image_inputs, return_tensors="pt", padding=True)
    labels = batch["input_ids"].clone()
    labels[labels == processor.tokenizer.pad_token_id] = -100

    # Mask image tokens in the labels
    image_tokens = [processor.tokenizer.convert_tokens_to_ids(processor.image_token)]
    for image_token_id in image_tokens:
        labels[labels == image_token_id] = -100
    batch["labels"] = labels

    return batch

def process_data(file_path):
    data = []
    with open(file_path, mode='r', encoding='utf-8') as csv_file:
        for row in csv.DictReader(csv_file):
            try:
                # Verify image exists
                image_path = row["image_path"]
                if not os.path.exists(image_path):
                    raise FileNotFoundError(f"Image not found: {image_path}")
                
                # Load and encode the image
                image = Image.open(image_path)
                
                # Prepare sample
                sample = {
                    "prompt": row["prompt"],
                    "image": image,
                    "response": row["response"],
                }
                
                # Format the data
                formatted_row = format_data(sample)
                data.append(formatted_row)
            except Exception as e:
                print(f"Error processing row {row['sl_no']}: {e}")
    return data

# Convert processed data into a Hugging Face Dataset
def create_dataset(data):
    dataset = Dataset.from_dict(
        {
            "messages": [entry["messages"] for entry in data]
        }
    )
    return dataset

import csv
import json

def convert_csv_to_json(csv_file_path, output_json_path):
    formatted_data = []
    with open(csv_file_path, mode="r", encoding="utf-8") as csv_file:
        reader = csv.DictReader(csv_file)
        for row in reader:
            dialogue = [
                {
                    "from": "human",
                    "value": f"<image>{row['prompt']}",
                },
                {
                    "from": "gpt",
                    "value": row["response"],
                },
            ]
            formatted_data.append({
                "dialogue": dialogue,
                "image_path": row["image_path"],
            })
    
    with open(output_json_path, "w", encoding="utf-8") as json_file:
        json.dump(formatted_data, json_file, indent=4)
    return output_json_path
