import csv
from PIL import Image
import json

# Define the formatting function
def format_data(sample):
    return {
        "messages": [
            {
                "role": "system",
                "content": [
                    {"type": "text", "text": "Please review the provided image and description."},
                ],
            },
            {
                "role": "user",
                "content": [
                    {
                        "type": "text",
                        "text": sample["prompt"],
                    },
                    {
                        "type": "image",
                        "image": sample["image"],
                    },
                ],
            },
            {
                "role": "assistant",
                "content": [
                    {"type": "text", "text": sample["response"]},
                ],
            },
        ],
    }

# Read the CSV and format the data
def process_csv(file_path):
    formatted_data = []
    with open(file_path, mode='r', encoding='utf-8') as csv_file:
        reader = csv.DictReader(csv_file)
        for row in reader:
            try:
                # Load the image
                image = Image.open(row["image_path"])
                # Format the sample
                sample = {
                    "prompt": row["prompt"],
                    "image": image,
                    "response": row["response"],
                }
                formatted_data.append(format_data(sample))
            except Exception as e:
                print(f"Error processing row {row['sl_no']}: {e}")
    return formatted_data

# Process and save the data as JSON for training
csv_file_path = "your_dataset.csv"  # Replace with your CSV file path
formatted_dataset = process_csv(csv_file_path)

# Save to JSON file
output_file = "formatted_dataset.json"
with open(output_file, mode='w', encoding='utf-8') as json_file:
    json.dump(formatted_dataset, json_file, indent=4)
    print(f"Formatted dataset saved to {output_file}")
