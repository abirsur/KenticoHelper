import json
from fuzzywuzzy import fuzz, process
import pandas as pd

# Sample lookup data

excel_file_path = 'book1.xlsx'
df = pd.read_excel(excel_file_path)
# Mapping of Excel fields to JSON fields
field_mapping = {
    "SortProducer Name": "Producer_Name",
    "SortProducerOne ID": "Producer_One_Id",
    "SortBilling Street": "Producer_Billing_Street",
    "SortBilling City": "Producer_Billing_City",
    "SortBilling State/Province(US & Canada)": "Producer_Billing_State",
    "SortBilling Country": "Producer_Billing_Country",
    "SortBilling Zip/Postal Code": "Producer_Billing_Zip_Postal_Code"
}

# Convert DataFrame to list of dictionaries and map fields
lookup_data = df.rename(columns=field_mapping).to_dict(orient='records')
    

def fuzzy_match(input_obj, lookup_data):
    best_match = None
    highest_score = 0
    
    for item in lookup_data:
        score = 0
        for key in input_obj:
            if input_obj[key] and item[key]:
                score += fuzz.ratio(input_obj[key], item[key])
        
        if score > highest_score:
            highest_score = score
            best_match = item
    
    return best_match

# Example input object
input_obj = {
    "Producer_Name": "Producer A",
    "Producer_Billing_Street": "123 Main Street, ste200",
    "Producer_Billing_City": "",
    "Producer_Billing_State": "",
    "Producer_Billing_Country": "USA",
    "Producer_Billing_Zip_Postal_Code": ""
}

# Perform fuzzy match
best_match = fuzzy_match(input_obj, lookup_data)
print(json.dumps(best_match, indent=4))
