import json
from fuzzywuzzy import fuzz

# Load data from JSON file
with open('extracted_data.json', 'r') as file:
    data = json.load(file)

# Filter data
filtered_data = [entry for entry in data if entry['type'].startswith('Acord') or entry['type'].startswith('Applied')]

# Separate into Client and Broker groups
client_data = [entry for entry in filtered_data if 'Client_Name' in entry['extracted_data']]
broker_data = [entry for entry in filtered_data if 'Broker_Agency_Name' in entry['extracted_data']]

# Function to perform fuzzy matching and fill missing data
def fuzzy_fill(data_list, key):
    for entry in data_list:
        if not entry['extracted_data'].get(key):
            best_match = None
            highest_score = 0
            for other_entry in data_list:
                if other_entry != entry and other_entry['extracted_data'].get(key):
                    score = fuzz.ratio(entry['extracted_data'].get(key, ''), other_entry['extracted_data'].get(key, ''))
                    if score > highest_score:
                        highest_score = score
                        best_match = other_entry
            if best_match:
                entry['extracted_data'][key] = best_match['extracted_data'][key]

# Perform fuzzy matching and fill missing data for non-client and non-broker data
for entry in filtered_data:
    if 'Client_Name' not in entry['extracted_data'] and 'Broker_Agency_Name' not in entry['extracted_data']:
        for key in entry['extracted_data']:
            fuzzy_fill(filtered_data, key)

# Prioritize Acord 125 if no matches are found
best_client_match = next((entry for entry in client_data if entry['type'] == 'Acord 125'), None)
best_broker_match = next((entry for entry in broker_data if entry['type'] == 'Acord 125'), None)

# Output the source of data with most valid data
output = {
    "best_client_match": best_client_match,
    "best_broker_match": best_broker_match
}

print(json.dumps(output, indent=4))
